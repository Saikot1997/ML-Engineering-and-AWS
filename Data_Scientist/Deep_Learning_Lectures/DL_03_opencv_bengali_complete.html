ইমেজ পড়া (imread) — Color vs Grayscale

ইমেজ পড়তে আমরা cv2-এর imread() function ব্যবহার করি। নাম দেখেই বোঝা যাচ্ছে — "image read" থেকে imread!

python imread ব্যবহার 
# রঙিন ছবি পড়া (default)
img_color = cv2.imread("street.jpg", cv2.IMREAD_COLOR)

# সাদাকালো ছবি পড়া
img_gray = cv2.imread("street.jpg", cv2.IMREAD_GRAYSCALE)

# type দেখো — দুটোই numpy array!
print('img_gray-এর type: ', type(img_gray))
print('img_color-এর type: ', type(img_color))

# Output:
# img_gray-এর type: <class 'numpy.ndarray'>
# img_color-এর type: <class 'numpy.ndarray'>

# Shape দেখো
print(img_color.shape)  # (150, 150, 3) — উচ্চতা, প্রস্থ, channel
print(img_gray.shape)   # (150, 150) — উচ্চতা, প্রস্থ
💡 .shape বুঝি সহজে

(150, 150, 3) মানে:

  • 150 → উচ্চতা (height) — উপর থেকে নিচে কতটি row আছে
  • 150 → প্রস্থ (width) — বাম থেকে ডানে কতটি column আছে
  • 3 → Color channel — Blue, Green, Red (BGR)

Grayscale-এ channel নেই, তাই শুধু (150, 150)।

চিত্র: imread-এর দুটি প্রধান mode
⚠️ সাবধান: imread শুধু কিছু format support করে
  • ✅ সাপোর্ট করে: .jpg, .jpeg, .png, .bmp, .tiff
  • ❌ সাপোর্ট করে না: .gif (animated), .webp (পুরোনো version-এ), .svg
  • ফাইল না পেলে None return করে — error দেয় না! তাই সবসময় check করো।

পিক্সেল কী? — ইমেজের গভীরে

একটা ছবিকে ভালো করে দেখলে বোঝা যাবে — এটা আসলে হাজারো ছোট ছোট রঙিন বর্গক্ষেত্রের সমষ্টি। প্রতিটি এই ছোট বর্গকে বলে Pixel (Picture Element সংক্ষেপে)।

Grayscale Pixel — সহজ উদাহরণ

Grayscale (সাদাকালো) ছবিতে প্রতিটি pixel হলো একটা সংখ্যা — 0 থেকে 255 পর্যন্ত।

চিত্র: একটা diagonal line-এর pixel মান। 255 = সাদা, 0 = কালো।

NumPy Array হিসেবে ইমেজ

python
# এই pixel grid তৈরি করো:
img_example = np.array([
    [255, 0, 0, 0, 0],
    [0, 255, 0, 0, 0],
    [0, 0, 255, 0, 0],
    [0, 0, 0, 255, 0],
    [0, 0, 0, 0, 255]
], dtype=np.uint8)  # uint8 = 0 থেকে 255 পর্যন্ত

print(img_example.shape)  # (5, 5)

# একটা নির্দিষ্ট pixel-এর মান দেখতে:
print(img_example[0, 0])  # 255 — প্রথম row, প্রথম column
print(img_example[2, 2])  # 255 — মাঝের pixel

# একটা pixel-এর মান বদলানো:
img_example[0, 0] = 128  # 128 = ধূসর রঙ
🎯 dtype=np.uint8 কেন?

uint8 মানে "unsigned 8-bit integer"। এটা মাত্র ৮ bit (1 byte) ব্যবহার করে এবং 0 থেকে 255 পর্যন্ত মান রাখতে পারে। ইমেজের প্রতিটি pixel 0-255 range-এ থাকে, তাই এটাই সঠিক datatype। float32 ব্যবহার করলে ৪ গুণ বেশি memory লাগবে!

ইমেজের উপর বাম দিকটা কালো করা

python
# উপর-বাম কোণের 75×75 pixel কালো করা
for i in range(75):
    for j in range(75):
        img_gray[i, j] = 0  # 0 মানে কালো

# আরও efficient উপায় (NumPy slicing):
img_gray[:75, :75] = 0  # একই কাজ, অনেক দ্রুত!
⚡ Production Tip: Loop বনাম NumPy Slicing

For loop ব্যবহার করলে ৭৫×৭৫ = ৫,৬২৫ বার iteration হয়। NumPy slicing একবারেই করে — প্রায় ১০০-১০০০ গুণ দ্রুত। সবসময় NumPy vectorization ব্যবহার করো।

💡 Mask কীভাবে কাজ করে — রান্নাঘরের উদাহরণ

pythonTriangle Mask তৈরি
# ধাপ ১: ছবির সাইজ নাও

# ধাপ ২: ত্রিভুজের তিনটা কোণের coordinate

plt.show()
চিত্র: Masking কীভাবে কাজ করে — bitwise AND operation

bitwise_and কীভাবে কাজ করে?

ধরো একটা pixel-এর মান ছবিতে 200, mask-এ 255 → AND করলে: 200 (রাখা হলো)।
আরেকটা pixel-এর মান ছবিতে 150, mask-এ 0 → AND করলে: 0 (বাদ গেলো)।

যেখানে mask = 255 (সাদা) → ছবির মূল মান রাখো। যেখানে mask = 0 (কালো) → কালো হয়ে যাবে।

ঘোরানো (Rotation)

python
def rotation(img, angle):
    """ছবিকে কোণে ঘোরানো"""
    long, large = img.shape[:2]  # height, width
    
    # ধাপ ১: Rotation Matrix তৈরি করো
    # args: center point, angle (degrees), scale
    M = cv2.getRotationMatrix2D(
        (int(long/2), int(large/2)),  # মাঝখান থেকে ঘোরাও
        angle,                          # কত ডিগ্রি?
        1                               # scale = 1 (সাইজ একই)
    )
    
    # ধাপ ২: Matrix apply করো
    img = cv2.warpAffine(img, M, (long, large))
    return img

# ব্যবহার:

Threshold Technique — Binary ও Adaptive

Threshold মানে হলো একটা সীমা নির্ধারণ করা। যেসব pixel সেই সীমার উপরে — তাদের সাদা (255) করো। নিচে — কালো (0) করো। এটাকে বলে Binarization — ছবিকে শুধু কালো ও সাদায় রূপান্তর।

import cv2
import numpy as np
from pathlib import Path
from typing import Optional, Tuple

def load_image(
    path: str,
    color: bool = True,
    target_size: Optional[Tuple[int, int]] = None,
    convert_rgb: bool = True
) -> Optional[np.ndarray]:
    """
    Production-ready image loading function.
    
    Args:
        path: ছবির পথ
        color: True=রঙিন, False=সাদাকালো
        target_size: (width, height) — resize করতে
        convert_rgb: BGR থেকে RGB convert করবে কিনা
    
    Returns:
        numpy array অথবা None (file না পেলে)
    """
    # File exist করে কিনা check
    if not Path(path).exists():
        print(f"Error: ফাইল পাওয়া যায়নি: {path}")
        return None
    
    # Flag নির্বাচন
    flag = cv2.IMREAD_COLOR if color else cv2.IMREAD_GRAYSCALE
    
    # ছবি পড়া
    img = cv2.imread(str(path), flag)
    
    # None check (corrupt file)
    if img is None:
        print(f"Error: ছবি পড়া যায়নি: {path}")
        return None
    
    # RGB convert (color image হলে)
    if color and convert_rgb:
        img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    
    # Resize
    if target_size is not None:
        img = cv2.resize(img, target_size, interpolation=cv2.INTER_AREA)
    
    return img

import random

def augment_image(img: np.ndarray, config: dict) -> np.ndarray:
    """
    ML project-এর জন্য augmentation pipeline।
    
    config উদাহরণ:
    {
        'rotate': True,     # ঘোরাবে?
        'rotate_range': 30, # কত ডিগ্রি পর্যন্ত (±)?
        'flip_h': True,     # horizontal flip?
        'flip_h_prob': 0.5, # কত probability?
        'flip_v': False,    # vertical flip?
        'zoom': True,       # zoom?
        'zoom_range': 0.1   # কতটুকু zoom (0-1)?
    }
    """
    result = img.copy()
    h, w = result.shape[:2]
    
    # ১. Random Rotation
    if config.get('rotate', False):
        angle = random.uniform(-config['rotate_range'], config['rotate_range'])
        M = cv2.getRotationMatrix2D((int(w/2), int(h/2)), angle, 1)
        result = cv2.warpAffine(result, M, (w, h))
    
    # ২. Horizontal Flip
    prob = config.get('flip_h_prob', 0.5)
    if config.get('flip_h', False) and random.random() < prob:
        result = cv2.flip(result, 1)
    
    # ৩. Vertical Flip
    if config.get('flip_v', False) and random.random() < 0.5:
        result = cv2.flip(result, 0)
    
    # ৪. Random Zoom (Crop)
    if config.get('zoom', False):
        zoom_f = random.uniform(0, config.get('zoom_range', 0.1))
        crop_h = int(h * zoom_f)
        crop_w = int(w * zoom_f)
        if crop_h > 0 and crop_w > 0:
            result = result[crop_h:h-crop_h, crop_w:w-crop_w]
            result = cv2.resize(result, (w, h))
    
    return result

def detect_edges(img: np.ndarray, method: str = 'canny') -> np.ndarray:
    """
    বিভিন্ন edge detection method।
    
    method:
        'canny'     → সবচেয়ে জনপ্রিয়, sharp edge
        'sobel'     → directional edge (x+y combined)
        'laplacian' → omnidirectional edge
    """
    # Grayscale convert করো
    if len(img.shape) == 3:
        gray = cv2.cvtColor(img, cv2.COLOR_RGB2GRAY)
    else:
        gray = img.copy()
    
    # Noise কমাও
    blurred = cv2.GaussianBlur(gray, (5, 5), 0)
    
    if method == 'canny':
        # Otsu দিয়ে threshold auto-select
        _, otsu = cv2.threshold(blurred, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
        high = int(np.max([otsu, blurred.mean() * 2]))
        low = int(high * 0.5)
        return cv2.Canny(blurred, low, high)
    
    elif method == 'sobel':
        sx = cv2.Sobel(blurred, cv2.CV_64F, 1, 0)
        sy = cv2.Sobel(blurred, cv2.CV_64F, 0, 1)
        magnitude = np.sqrt(sx**2 + sy**2)
        return np.uint8(np.clip(magnitude, 0, 255))
    
    elif method == 'laplacian':
        lap = cv2.Laplacian(blurred, cv2.CV_64F)
        return np.uint8(np.abs(lap))
    
    else:
        raise ValueError(f"অজানা method: {method}")

# ব্যবহার:

def binarize(img: np.ndarray, method: str = 'otsu') -> np.ndarray:
    """
    ছবিকে binary (কালো-সাদা) করা।
    
    method:
        'otsu'     → auto threshold (সবচেয়ে ভালো general)
        'adaptive' → variable threshold (uneven lighting)
        'manual'   → নিজে threshold দাও
    """
    if len(img.shape) == 3:
        gray = cv2.cvtColor(img, cv2.COLOR_RGB2GRAY)
    else:
        gray = img
    
    blurred = cv2.GaussianBlur(gray, (3, 3), 0)
    
    if method == 'otsu':
        # Otsu = সবচেয়ে ভালো threshold auto খোঁজে
        _, binary = cv2.threshold(blurred, 0, 255,
                                   cv2.THRESH_BINARY + cv2.THRESH_OTSU)
        return binary
    
    elif method == 'adaptive':
        return cv2.adaptiveThreshold(
            blurred, 255,
            cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
            cv2.THRESH_BINARY,
            blockSize=11, C=5
        )
    
    elif method == 'manual':
        _, binary = cv2.threshold(blurred, 127, 255, cv2.THRESH_BINARY)
        return binary

06 Morphological Processing Morphology

🔬 সহজ ব্যাখ্যা: ছবির আকার/গঠন পরিবর্তন। ছোট করা (erosion), বড় করা (dilation), এবং তাদের সমন্বয়।

pythonMorphological Operations
def morphological_process(
    binary_img: np.ndarray,
    operation: str = 'opening',
    kernel_size: int = 5
) -> np.ndarray:
    """
    Morphological operation।
    
    operation:
        'erosion'  → ছোট করা (noise দূর)
        'dilation' → বড় করা (hole fill)
        'opening'  → erosion তারপর dilation (noise remove)
        'closing'  → dilation তারপর erosion (hole fill)
        'gradient' → contour বের করা
    """
    kernel = np.ones((kernel_size, kernel_size), np.uint8)
    
    ops = {
        'erosion' : lambda img: cv2.erode(img, kernel),
        'dilation': lambda img: cv2.dilate(img, kernel),
        'opening' : lambda img: cv2.morphologyEx(img, cv2.MORPH_OPEN,     kernel),
        'closing' : lambda img: cv2.morphologyEx(img, cv2.MORPH_CLOSE,    kernel),
        'gradient': lambda img: cv2.morphologyEx(img, cv2.MORPH_GRADIENT, kernel),
    }
    
    if operation not in ops:
        raise ValueError(f"অজানা operation: {operation}")
    
    return ops[operation](binary_img)

# ব্যবহার — noise দূর করা:

    # ধাপ ৫: Normalize (0-255 → 0.0-1.0)

def create_polygon_mask(
    img_shape: Tuple[int, int],
    vertices: List[Tuple[int, int]]
) -> np.ndarray:
    """
    যেকোনো polygon আকারের mask তৈরি করো।
    
    Args:
        img_shape: (height, width)
        vertices: [(x1,y1), (x2,y2), ...] — polygon-এর কোণগুলো
    
    Returns:
        Binary mask array
    """
    mask = np.zeros(img_shape, dtype=np.uint8)
    pts = np.array([vertices], dtype=np.int32)
    cv2.fillPoly(mask, pts, color=255)
    return mask

def apply_roi_mask(
    img: np.ndarray,
    mask_type: str = 'triangle_bottom',
    custom_vertices: List = None
) -> np.ndarray:
    """
    Pre-defined ROI mask apply করো।
    
    mask_type:
        'triangle_bottom' → নিচের ত্রিভুজ (lane detection)
        'rectangle_center'→ মাঝের আয়তক্ষেত্র (product inspection)
        'custom'          → নিজের polygon দাও
    """
    h, w = img.shape[:2]
    gray_shape = (h, w)

    if mask_type == 'triangle_bottom':
        vertices = [(w//2, h//2), (0, h), (w, h)]
    elif mask_type == 'rectangle_center':
        margin_x, margin_y = w//5, h//5
        vertices = [
            (margin_x, margin_y), (w-margin_x, margin_y),
            (w-margin_x, h-margin_y), (margin_x, h-margin_y)
        ]
    elif mask_type == 'custom':
        vertices = custom_vertices
    else:
        raise ValueError(f"অজানা mask_type: {mask_type}")

    mask = create_polygon_mask(gray_shape, vertices)

    # Color image-এ mask apply
    if len(img.shape) == 3:
        mask_3ch = np.stack([mask]*3, axis=-1)
        return cv2.bitwise_and(img, mask_3ch)
    return cv2.bitwise_and(img, mask)

def find_and_draw_contours(
    img: np.ndarray,
    min_area: int = 100,
    draw_bbox: bool = True
) -> Tuple[np.ndarray, int]:
    """
    ছবিতে সমস্ত object-এর contour খোঁজো এবং আঁকো।
    
    Args:
        img: input image (color বা grayscale)
        min_area: সর্বনিম্ন area (ছোট noise বাদ দিতে)
        draw_bbox: bounding box আঁকবে?
    
    Returns:
        (annotated_image, object_count)
    """
    result = img.copy()

    # Grayscale ও binary করো
    if len(img.shape) == 3:
        gray = cv2.cvtColor(img, cv2.COLOR_RGB2GRAY)
    else:
        gray = img

    blurred = cv2.GaussianBlur(gray, (5, 5), 0)
    _, binary = cv2.threshold(blurred, 0, 255,
                               cv2.THRESH_BINARY + cv2.THRESH_OTSU)

    # Contour খোঁজো
    contours, _ = cv2.findContours(
        binary, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE
    )

    # ছোট noise বাদ দাও
    significant = [c for c in contours if cv2.contourArea(c) > min_area]

    for i, cnt in enumerate(significant):
        # Contour আঁকো (সবুজ)
        cv2.drawContours(result, [cnt], -1, (0, 255, 0), 2)

        if draw_bbox:
            # Bounding box (নীল)
            x, y, w, h = cv2.boundingRect(cnt)
            cv2.rectangle(result, (x, y), (x+w, y+h), (255, 0, 0), 2)
            # Number label
            cv2.putText(result, str(i+1), (x, y-10),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.7, (255, 255, 0), 2)

    return result, len(significant)

class ImageProcessor:
    """
    ML Project-এর জন্য সম্পূর্ণ Image Processing class।
    
    ব্যবহার:
        processor = ImageProcessor(target_size=(224, 224))
        img = processor.load("image.jpg")
        img = processor.preprocess(img, denoise=True)
        img = processor.augment(img)
        tensor = processor.to_tensor(img)
    """
    
    def __init__(
        self,
        target_size: Tuple[int, int] = (224, 224),
        color_mode: str = 'rgb'
    ):
        self.target_size = target_size
        self.color_mode = color_mode
        self.processing_log = []

    def load(self, path: str) -> Optional[np.ndarray]:
        """ছবি load করো সব safety check সহ।"""
        img = cv2.imread(path)
        if img is None:
            self.processing_log.append(f"FAIL: {path}")
            return None
        if self.color_mode == 'rgb':
            img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
        elif self.color_mode == 'gray':
            img = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        self.processing_log.append(f"OK: {path} {img.shape}")
        return img

    def preprocess(
        self,
        img: np.ndarray,
        denoise: bool = True,
        equalize: bool = False
    ) -> np.ndarray:
        """Standard preprocessing।"""
        # Resize
        img = cv2.resize(img, self.target_size, interpolation=cv2.INTER_AREA)
        # Denoise
        if denoise:
            if len(img.shape) == 3:
                img = cv2.GaussianBlur(img, (3, 3), 0)
            else:
                img = cv2.GaussianBlur(img, (3, 3), 0)
        # Histogram Equalization (contrast বাড়ানো)
        if equalize and len(img.shape) == 2:
            img = cv2.equalizeHist(img)
        return img

    def augment(
        self,
        img: np.ndarray,
        rotate_range: float = 20,
        flip_prob: float = 0.5
    ) -> np.ndarray:
        """Random augmentation।"""
        h, w = img.shape[:2]
        # Random rotation
        angle = random.uniform(-rotate_range, rotate_range)
        M = cv2.getRotationMatrix2D((int(w/2), int(h/2)), angle, 1)
        img = cv2.warpAffine(img, M, (w, h))
        # Random flip
        if random.random() < flip_prob:
            img = cv2.flip(img, 1)
        return img

    def to_tensor(self, img: np.ndarray) -> np.ndarray:
        """ML model-এর জন্য normalize ও reshape।"""
        tensor = img.astype(np.float32) / 255.0
        if len(tensor.shape) == 2:
            tensor = tensor[..., np.newaxis]  # (H,W) → (H,W,1)
        return tensor

    def detect_edges(self, img: np.ndarray) -> np.ndarray:
        """Canny edge detection।"""
        if len(img.shape) == 3:
            gray = cv2.cvtColor(img, cv2.COLOR_RGB2GRAY)
        else:
            gray = img
        blurred = cv2.GaussianBlur(gray, (5, 5), 0)
        return cv2.Canny(blurred, 50, 150)

    def get_log(self) -> List[str]:
        """Processing log দেখো।"""
        return self.processing_log
