tf.constant — কনস্ট্যান্ট টেনসর তৈরি করা

tf.constant() দিয়ে এমন টেনসর তৈরি করা হয় যার মান পরে পরিবর্তন করা যাবে না — ঠিক পাথরে খোদাই করা লেখার মতো।

প্যারামিটার ব্যাখ্যা
value: আসল মান (সংখ্যা, লিস্ট, অ্যারে)
dtype: ডেটার ধরন — tf.float32, tf.int32 ইত্যাদি
shape: আকার (ঐচ্ছিক) — যেমন [3,4] মানে ৩ সারি ৪ কলাম
name: নাম (ঐচ্ছিক) — গ্রাফে খুঁজে পেতে সাহায্য করে
Python — tf.constant ব্যবহার (Notebook 1, Exercise 1) 
import tensorflow as tf

# ধাপ ১: দুটি কনস্ট্যান্ট টেনসর তৈরি করো
x1 = tf.constant(5)    # একটি স্কেলার — মান ৫
x2 = tf.constant(10)   # একটি স্কেলার — মান ১০

print(x1)  # tf.Tensor(5, shape=(), dtype=int32)
print(x2)  # tf.Tensor(10, shape=(), dtype=int32)

# লক্ষ্য করো: shape=() মানে 0D (স্কেলার)
# dtype=int32 মানে পূর্ণসংখ্যা

# ধাপ ২: যোগ করো
result = tf.add(x1, x2)   # অথবা: result = x1 + x2
print(result)              # tf.Tensor(15, shape=(), dtype=int32)

# NumPy মান বের করতে .numpy() ব্যবহার করো
print(result.numpy())     # 15 (Python int হিসেবে)
⚠️ গুরুত্বপূর্ণ: print(x1) করলে "৫" দেখায় না, দেখায় tf.Tensor(5, shape=(), dtype=int32)। আসল মান দেখতে x1.numpy() ব্যবহার করতে হবে।

কল্পনা করো একটি বর্গ মাঠের ভেতরে একটি গোল পুল (quarter circle)। তুমি চোখ বন্ধ করে এলোমেলো পাথর ছুড়ছ। কতগুলো পাথর পুলে পড়ল সেটা গণনা করলে π এর আনুমানিক মান বের করা যায়!

import tensorflow as tf

# ধাপ ১: ১০,০০০টি এলোমেলো বিন্দু তৈরি করো
# প্রতিটি বিন্দুর x এবং y স্থানাঙ্ক ০ থেকে ১ এর মধ্যে
X = tf.random.uniform([10000, 2])

print('Min of X :', tf.reduce_min(X).numpy())  # ≈ 0.0
print('Max of X :', tf.reduce_max(X).numpy())  # ≈ 1.0

# ধাপ ২: কোন বিন্দুগুলো বৃত্তের ভেতরে?
# শর্ত: x² + y² < 1 হলে বৃত্তের ভেতরে

# পদ্ধতি ১: for loop (ধীরে কিন্তু বোঝা সহজ)
a = 0
for i in range(len(X)):
    if X[i, 0]**2 + X[i, 1]**2 < 1:
        a += 1

# পদ্ধতি ২: ভেক্টরাইজড (দ্রুত, ML এ এইভাবে করা হয়)
mask = X[:, 0]**2 + X[:, 1]**2 < 1
a = tf.reduce_sum(tf.cast(mask, tf.int32)).numpy()

print(f'বৃত্তের ভেতরে: {a} টি বিন্দু')
print(f'π এর আনুমান: {a / 10000 * 4:.4f}')  # ≈ 3.1416

constant vs variable
tf.constant = পাথরে খোদাই — একবার লিখলে আর বদলানো যাবে না
tf.Variable = ব্ল্যাকবোর্ডে লেখা — যতবার খুশি মুছে আবার লেখা যাবে

Neural Network-এর weights হলো Variable — training-এ প্রতিটি ধাপে এগুলো আপডেট হয়।
tf.Variable ব্যবহার
import tensorflow as tf

# Variable তৈরি — প্রাথমিক মান 3.0
x = tf.Variable(initial_value=3.0)

# ফাংশন সংজ্ঞা: f(x) = x² - 2x + 3
def f(x):
    return x*x - 2*x + 3

print(f'f(3) = {f(x).numpy()}')  # 9 - 6 + 3 = 6

# মান পরিবর্তন করো
x.assign(4.0)
print(f'f(4) = {f(x).numpy()}')  # 16 - 8 + 3 = 11
GradientTape — স্বয়ংক্রিয় ডেরিভেটিভ
গ্র্যাডিয়েন্ট কী? — সহজ বাংলায়
ধরো তুমি পাহাড়ে আছ এবং নিচে নামতে চাও (minimum খুঁজছ)। গ্র্যাডিয়েন্ট তোমাকে বলে কোন দিকে ঢাল আছে এবং ঢাল কতটা খাড়া। তুমি ঢালের বিপরীত দিকে পা ফেলবে — এটাই Gradient Descent!
f(x) = x² - 2x + 3 এর গ্রাফ এবং Gradient Descent
GradientTape — গ্র্যাডিয়েন্ট বের করা
# GradientTape বোঝার সহজ উদাহরণ
# f(x) = x² এর ডেরিভেটিভ x=3 তে = 2x = 6

x = tf.Variable(3.0)

with tf.GradientTape() as tape:
    y = x * x  # tape সব operation রেকর্ড করে

grad = tape.gradient(y, x)  # dy/dx = 2x = 6
print(f'gradient: {grad.numpy()}')  # 6.0

# f(x) = x² - 2x + 3 এর জন্য
# df/dx = 2x - 2, x=4 তে = 2(4)-2 = 6
x.assign(4.0)
with tf.GradientTape() as tape:
    function = f(x)

grad = tape.gradient(function, x)
print(f'df/dx at x=4: {grad.numpy()}')  # 6.0
print(f'Analytical: {2*4-2}')             # 6 ✓
⚠️ গুরুত্বপূর্ণ নিয়ম: GradientTape শুধুমাত্র tf.Variable-এর gradient track করে, tf.constant-এর নয়। Neural Network-এ weights সবসময় Variable হয় এই কারণেই।

        function = f(x)          # forward pass: মান বের করো
    
    grad = tape.gradient(function, x)  # গ্র্যাডিয়েন্ট বের করো
    
    # Update rule: x = x - learning_rate × gradient
    x.assign(x - learning_rate * grad)

import tensorflow as tf
from sklearn.datasets import make_regression

# ─── Model সংজ্ঞা ───
class Linear():
    def __init__(self):
        # w এবং b — এগুলো আমরা optimize করব
        self.w = tf.Variable(tf.random.normal([1]), name='weight')
        self.b = tf.Variable(tf.random.normal([1]), name='bias')
    
    def __call__(self, inputs):
        # ŷ = b + w × x
        return inputs * self.w + self.b

# ─── ডেটা তৈরি ───
X, y = make_regression(n_samples=100, n_features=1, noise=10, random_state=0)
X = X.squeeze()  # shape (100,1) → (100,)

# ─── Training ───
model = Linear()
optimizer = tf.keras.optimizers.Adam(learning_rate=0.1)

def train_op(model, inputs, targets):
    with tf.GradientTape() as tape:
        y_pred = model(inputs)    # forward pass
        loss = tf.keras.losses.mean_squared_error(targets, y_pred)
    
    grads = tape.gradient(loss, [model.w, model.b])  # w ও b-এর gradient
    optimizer.apply_gradients(zip(grads, [model.w, model.b]))
    return loss.numpy()

# ১০০০ ধাপে train করো
losses = [train_op(model, X, y) for _ in range(1000)]

print(f'w = {model.w.numpy()[0]:.2f}')   # ≈ 40 (slope)
print(f'b = {model.b.numpy()[0]:.2f}')   # ≈ 0  (intercept)

class Polynomial(tf.keras.Model):
    def __init__(self):
        super(Polynomial, self).__init__()  # অবশ্যই করতে হবে
        self.w0 = tf.Variable(tf.random.normal([1]), name='w0')
        self.w1 = tf.Variable(tf.random.normal([1]), name='w1')
        self.w2 = tf.Variable(tf.random.normal([1]), name='w2')
        self.b  = tf.Variable(tf.random.normal([1]), name='bias')
    
    def __call__(self, inputs, training=True):
        # ŷ = b + w0×x + w1×x² + w2×x³
        return self.b + self.w0*inputs + self.w1*inputs**2 + self.w2*inputs**3

def transform(x, y):
    img = tf.reshape(x, (28, 28, 1))       # সঠিক shape
    img = tf.image.random_crop(img, [25, 25, 1])  # random crop (augmentation)
    return tf.image.resize(img, (28, 28)), y   # আবার resize

# ৩. Pipeline chain
dataset = (dataset
    .map(transform, num_parallel_calls=tf.data.AUTOTUNE)  # প্রতিটিতে transform
    .shuffle(buffer_size=20000)                           # এলোমেলো করো
    .batch(64)                                             # ৬৪টি করে batch
    .prefetch(tf.data.AUTOTUNE)                            # আগে থেকে load করো

Custom Training Loop — CNN
# Custom loss function ও optimizer
optimizer = tf.keras.optimizers.Adam(learning_rate=0.001)

def train_op(model, inputs, targets):
    with tf.GradientTape() as tape:
        y_pred = model(inputs, training=True)  # Dropout চালু থাকবে
        loss_value = tf.reduce_mean(
            tf.keras.losses.sparse_categorical_crossentropy(targets, y_pred)
        )
    # trainable_variables: সব learnable parameter (Conv filters, Dense weights)
    grads = tape.gradient(loss_value, model.trainable_variables)
    optimizer.apply_gradients(zip(grads, model.trainable_variables))
    return loss_value.numpy()

# ৫ epoch training

        tf.keras.losses.sparse_categorical_crossentropy(
            y_test, lenet(X_test.reshape([-1, 28, 28, 1]))

সহজ উপমা
ধরো তুমি রান্না করছ। Dense Layer হলো সুপারশপের ready-made সস। কিন্তু তোমার দাদির বিশেষ রেসিপি দরকার? তাহলে নিজেই তৈরি করতে হবে — সেটাই Custom Layer। Keras-এর Layer class-কে inherit করে নিজের layer বানানো যায়, এবং এটি keras-এর সব সুবিধা পায়।
Custom Layer-এর কাঠামো
তিনটি মূল Method — প্রতিটি কী করে?
__init__ Method
কাজ: Layer তৈরি হওয়ার সময় প্রথমে এটি চলে। Hyperparameter (যেমন units) সংরক্ষণ করা হয়। Parent class-এর __init__ অবশ্যই call করতে হয় — super().__init__() দিয়ে।

উপমা: কারখানার ব্লুপ্রিন্ট তৈরি — কতটা বড় হবে সেটা এখানে ঠিক করা হয়।
build Method
কাজ: প্রথমবার data দিলে weight তৈরি হয়। input_shape argument থেকে input-এর আকার জেনে সঠিক size-এর weight বানানো হয়। add_weight() method দিয়ে trainable weight যোগ করা হয়।

উপমা: কারখানা চালু হওয়ার সময় মেশিন বসানো।
call Method
কাজ: প্রতিবার data দিলে এটি চলে। Input নিয়ে গাণিতিক operation করে output বের করে। এটাই layer-এর মূল কাজ — forward pass।

উপমা: কারখানায় প্রতিদিনের উৎপাদন — কাঁচামাল ঢোকে, পণ্য বের হয়।
উদাহরণ ১ — CustomLinear Layer (সহজতম)

প্রথমে সবচেয়ে সহজ example — একটি Linear layer যেটা শুধু Y = XW + b করে।

CustomLinear Layer
import tensorflow as tf
from tensorflow.keras.layers import Layer

class CustomLinear(Layer):

    def __init__(self, units):
        super(CustomLinear, self).__init__()  # Parent-এর __init__ চালাও
        self.units = units                       # কতটি neuron

    def build(self, input_shape):
        self.w = self.add_weight(
            shape=(input_shape[-1], self.units),  # (features, units)
            initializer="random_normal",
            trainable=True
        )
        self.b = self.add_weight(
            shape=(self.units,),
            initializer="random_normal",
            trainable=True
        )

    def call(self, inputs):
        return tf.matmul(inputs, self.w) + self.b   # Y = XW + b

# ─── ব্যবহার ───
linear = CustomLinear(5)         # 5 neuron-এর layer
data = tf.ones((3, 4))            # 3 sample, 4 feature
output = linear(data)             # প্রথম call-এ build() স্বয়ংক্রিয়ভাবে চলে
print(output.shape)               # (3, 5)
⚠️ গুরুত্বপূর্ণ: build() কখনো manually call করতে হয় না। প্রথমবার data দিলে Keras স্বয়ংক্রিয়ভাবে build() call করে এবং weight তৈরি করে।
উদাহরণ ২ — CustomPolynomial Layer (দ্বিতীয় ক্রম)
ধারণা
Linear regression: y = w₁x + b
দ্বিতীয় ক্রম polynomial: y = w₁x + w₂x² + b

এজন্য input [x] কে [x, x²] করতে হবে — concatenate করে দ্বিগুণ। তাই weight-এর shape হবে (features×2, units)।
Polynomial Layer-এর data প্রবাহ
CustomPolynomial Layer — সম্পূর্ণ কোড
class CustomPolynomial(Layer):

    def __init__(self, units):
        super(CustomPolynomial, self).__init__()
        self.units = units

    def build(self, input_shape):
        # ×2 কারণ আমরা x এবং x² দুটোই দেব
        self.w = self.add_weight(
            shape=(input_shape[-1] * 2, self.units),
            initializer="random_normal",
            trainable=True
        )
        self.b = self.add_weight(
            shape=(self.units,),
            initializer="random_normal",
            trainable=True
        )

    def call(self, inputs):
        inputs_pow = tf.pow(inputs, 2)
        inputs_process = tf.concat([inputs, inputs_pow], axis=-1)
        return tf.matmul(inputs_process, self.w) + self.b

# ─── পরীক্ষা ───
customlayer = CustomPolynomial(8)
inputs = tf.ones((4, 5))
output = customlayer(inputs)
print(output.shape)   # (4, 8)

# Weight সম্পর্কে তথ্য
print("মোট weight:",        len(customlayer.weights))
print("Trainable weight:",  len(customlayer.trainable_weights))
print("Non-trainable:",     len(customlayer.non_trainable_weights))
✅ মনে রাখো: weights, trainable_weights, non_trainable_weights attribute দিয়ে weight-এর তথ্য জানা যায়। BatchNormalization-এ non-trainable weight থাকে (moving mean/variance)।

class CustomDropout(Layer):

    def __init__(self, rate):
        super(CustomDropout, self).__init__()
        self.rate = rate   # e.g. 0.2 = 20% neuron বন্ধ

    def call(self, inputs, training=None):
        if training:
            return tf.nn.dropout(inputs, rate=self.rate)
        return inputs  # inference-এ কিছুই করে না

class L2Normalization(Layer):
    def __init__(self):
        super().__init__()

    def call(self, inputs):
        norm = tf.norm(inputs, ord=2, axis=-1, keepdims=True)
        return inputs / (norm + 1e-8)  # epsilon যোগ করি zero-division এড়াতে

class ResidualBlock(Layer):
    def __init__(self, units):
        super().__init__()
        self.dense1 = Dense(units, activation='relu')
        self.dense2 = Dense(units)   # activation নেই — add করার আগে

    def call(self, inputs, training=None):
        x = self.dense1(inputs)
        x = self.dense2(x)
        return tf.nn.relu(x + inputs)  # Skip connection!

class LossHistory(callbacks.Callback):
    def on_train_begin(self, logs=None):
        self.losses = []

    def on_epoch_end(self, epoch, logs=None):
        self.losses.append({
            'epoch': epoch,
            'train_loss': logs.get('loss'),
            'val_loss': logs.get('val_loss')
        })

    def on_train_end(self, logs=None):
        print("=== Loss History ===")
        for e in self.losses:
            print(f"Epoch {e['epoch']}: train={e['train_loss']:.4f}, val={e['val_loss']:.4f}")

history_cb = LossHistory()
model.fit(X_train, y_train, epochs=5, validation_split=0.2,
         callbacks=[history_cb])

প্র. ১
tf.constant এবং tf.Variable-এর পার্থক্য কী? কোথায় কোনটা ব্যবহার করবে?
উত্তর: tf.constant immutable — একবার তৈরি হলে মান বদলানো যায় না। tf.Variable mutable — .assign() দিয়ে মান পরিবর্তন করা যায়। Neural Network weights সবসময় Variable কারণ training-এ এগুলো আপডেট করতে হয়। Input data সাধারণত constant। GradientTape শুধু Variable-এর gradient track করে।
প্র. ২
GradientTape কীভাবে কাজ করে? কেন এটা ব্যবহার করা হয়?
উত্তর: GradientTape একটি "context manager" যেটা with block-এর ভেতরে সব operation রেকর্ড করে। তারপর .gradient(output, variable) call করলে chain rule ব্যবহার করে automatic differentiation করে। এটা backpropagation-এর ভিত্তি — manually derivative বের করতে হয় না।
প্র. ৩
Overfitting কী? কীভাবে detect ও prevent করবে?
উত্তর: Training accuracy অনেক বেশি কিন্তু Validation accuracy কম — এটাই overfitting। Model training data মুখস্থ করে ফেলেছে, নতুন data-তে কাজ করছে না।

Detect: Training vs Validation loss/accuracy graph দেখো। Gap বড় হলে overfitting।
Prevent: ① Dropout ② L1/L2 Regularization ③ Data Augmentation ④ Early Stopping ⑤ কম parameter-এর model ⑥ বেশি training data
প্র. ৪
CNN-এর Conv2D layer-এ "padding='same'" কী করে এবং কেন দরকার?
উত্তর: padding='same' ছবির চারপাশে শূন্য (zero) যোগ করে যাতে output-এর spatial dimension input-এর মতোই থাকে। padding='valid' হলে output ছোট হয়ে যায় (প্রতিটি Conv layer-এ)। Deep network-এ 'same' ব্যবহার করলে feature map-এর আকার নিয়ন্ত্রণ করা সহজ হয়।
প্র. ৫
Batch size কীভাবে model performance-এ প্রভাব ফেলে?
উত্তর:
• ছোট batch (8-32): noisy gradient, regularization effect, generalization ভালো, কিন্তু ধীর convergence
• বড় batch (256-1024): stable gradient, দ্রুত training (GPU efficient), কিন্তু overfitting ঝুঁকি বেশি, sharp minima-তে আটকে যেতে পারে
• সাধারণ সুপারিশ: ৩২ বা ৬৪ দিয়ে শুরু করো
প্র. ৬
Transfer Learning কী এবং কখন ব্যবহার করবে?
উত্তর: অন্য বড় dataset-এ (যেমন ImageNet) train করা model-এর weights নিজের কাজে ব্যবহার করা। কারণ low-level features (edges, textures) সব ছবিতে একই রকম। ব্যবহার করবে যখন: ① নিজের dataset ছোট ② compute কম ③ দ্রুত prototype দরকার। Pre-trained model freeze করে শুধু শেষের layers train করলে ভালো ফলাফল পাওয়া যায়।
প্র. ৭
tf.data.Dataset pipeline-এ prefetch() কেন ব্যবহার করো?
উত্তর: GPU model চালানোর সময় CPU পরবর্তী batch prepare করে রাখে। এতে GPU কখনো idle থাকে না — training দ্রুত হয়। tf.data.AUTOTUNE দিলে TensorFlow নিজে optimal buffer size ঠিক করে।
প্র. ৮
Softmax vs Sigmoid activation — কোথায় কোনটা?
উত্তর:
• Sigmoid: Binary classification-এ (output layer-এ ১টি neuron), probability ০-১ এর মধ্যে, independent
• Softmax: Multi-class classification-এ (output layer-এ k টি neuron), সব outputs মিলিয়ে ১ হয়, mutual exclusive
• Rule: শুধু হ্যাঁ/না → Sigmoid; একাধিক class-এর মধ্যে একটি → Softmax
