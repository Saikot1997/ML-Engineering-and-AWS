একটি function লিখো যা একটি URL নিয়ে সেই পেজের সমস্ত paragraph text return করে, এবং যদি request fail করে তাহলে gracefully handle করে।

Python — ML Engineering স্টাইল
import requests
from bs4 import BeautifulSoup
import logging
from typing import List, Optional

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

def extract_paragraphs(url: str, timeout: int = 10) -> Optional[List[str]]:
    """
    একটি URL থেকে সব paragraph text বের করে।
    
    Args:
        url: স্ক্র্যাপ করার URL
        timeout: Request timeout (seconds)
    
    Returns:
        List of paragraph strings, অথবা None যদি fail করে
    """
    try:
        headers = {
            'User-Agent': 'Mozilla/5.0 (research bot)'
        }
        response = requests.get(url, timeout=timeout, headers=headers)
        response.raise_for_status()  # 4xx/5xx error হলে exception raise করে
        
        soup = BeautifulSoup(response.content, 'lxml')
        paragraphs = [
            p.get_text(strip=True)
            for p in soup.find_all('p')
            if p.get_text(strip=True)  # খালি paragraph বাদ
        ]
        logger.info(f"Found {len(paragraphs)} paragraphs from {url}")
        return paragraphs
        
    except requests.Timeout:
        logger.error(f"Timeout: {url} respond করেনি {timeout}s-এর মধ্যে")
    except requests.HTTPError as e:
        logger.error(f"HTTP Error {e.response.status_code}: {url}")
    except Exception as e:
        logger.error(f"Unexpected error: {e}")
    
    return None

কোডিং প্রশ্ন (ML Engineering)

একটি e-commerce সাইটে প্রতিটি product card এভাবে আছে:
<div class="product-card" data-id="123"><span class="price">৳500</span></div>

Python Solution
import pandas as pd
from bs4 import BeautifulSoup

def extract_products(html_content: str) -> pd.DataFrame:
    soup = BeautifulSoup(html_content, 'lxml')
    products = []
    
    for card in soup.find_all('div', class_='product-card'):
        product_id = card.get('data-id')  # data attribute বের করা
        price_tag = card.find('span', class_='price')
        
        # None check — সবসময় করতে হবে!
        if price_tag is not None:
            price_text = price_tag.get_text(strip=True)
            # ৳ সরিয়ে number বের করা
            price = float(price_text.replace('৳', '').replace(',', '').strip())
        else:
            price = None
        
        products.append({'product_id': product_id, 'price': price})
    
    return pd.DataFrame(products)

Python — Production Grade Scraper
import requests, time, logging
from typing import Optional

logger = logging.getLogger(__name__)

def safe_get(
    url: str,
    max_retries: int = 3,
    backoff: float = 2.0,
    timeout: int = 15,
    headers: Optional[dict] = None
) -> Optional[requests.Response]:
    """
    Production-grade HTTP GET with exponential backoff retry।
    
    Exponential Backoff মানে: প্রথম try fail → 2s অপেক্ষা
    দ্বিতীয় fail → 4s অপেক্ষা, তৃতীয় fail → 8s অপেক্ষা।
    """
    _headers = headers or {
        'User-Agent': 'Mozilla/5.0 (Research Bot)'
    }
    
    for attempt in range(max_retries):
        try:
            response = requests.get(url, headers=_headers, timeout=timeout)
            
            if response.status_code == 429:  # Rate limited
                wait = backoff ** attempt
                logger.warning(f"Rate limited. Waiting {wait}s...")
                time.sleep(wait)
                continue
            
            response.raise_for_status()
            time.sleep(0.5)  #礼貌的 delay — সাইটকে চাপ না দেওয়া
            return response
            
        except requests.Timeout:
            logger.warning(f"Timeout on attempt {attempt+1}/{max_retries}")
        except requests.HTTPError as e:
            logger.error(f"HTTP {e.response.status_code} for {url}")
            if e.response.status_code == 404:
                break  # 404-এ retry করা মানে নেই
        
        time.sleep(backoff ** attempt)  # Exponential backoff
    
    return None

import requests, time, pandas as pd
from bs4 import BeautifulSoup
from typing import List, Dict

def scrape_books(base_url: str, max_pages: int = 5) -> pd.DataFrame:
    """
    books.toscrape.com থেকে বই scrape করার উদাহরণ।
    এটা legal scraping practice site।
    """
    all_books: List[Dict] = []
    
    for page in range(1, max_pages + 1):
        url = f"{base_url}/catalogue/page-{page}.html"
        response = requests.get(url, timeout=10)
        
        if response.status_code != 200:
            print(f"Page {page} failed: {response.status_code}")
            break
        
        soup = BeautifulSoup(response.content, 'lxml')
        
        for article in soup.find_all('article', class_='product_pod'):
            title = article.find('h3').find('a')['title']
            price = article.find('p', class_='price_color').text.strip()
            rating_class = article.find('p', class_='star-rating')['class'][1]
            
            # Rating শব্দকে সংখ্যায় রূপান্তর
            rating_map = {'One':1,'Two':2,'Three':3,'Four':4,'Five':5}
            rating = rating_map.get(rating_class, 0)
            
            all_books.append({
                'title': title,
                'price': float(price.replace('Â£','').replace('£','')),
                'rating': rating,
                'page': page
            })
        
        time.sleep(1)  #礼貌: সাইটকে চাপ না দেওয়া
        print(f"Page {page} done: {len(all_books)} books total")
    
    df = pd.DataFrame(all_books)
    df.to_csv('books_dataset.csv', index=False)
    return df

# quotes.toscrape.com — practice site, legal
URL = "https://quotes.toscrape.com"

def scrape_all_quotes() -> pd.DataFrame:
    quotes_data = []
    page_url = URL
    
    while page_url:  # যতক্ষণ পরের page আছে
        soup = BeautifulSoup(
            requests.get(page_url).content, 'lxml'
        )
        
        for quote in soup.find_all('div', class_='quote'):
            text   = quote.find('span', class_='text').get_text(strip=True)
            author = quote.find('small', class_='author').get_text(strip=True)
            tags   = [t.get_text() for t in quote.find_all('a', class_='tag')]
            
            quotes_data.append({
                'text': text,
                'author': author,
                'tags': ', '.join(tags)
            })
        
        # পরের page আছে কিনা দেখো
        next_btn = soup.find('li', class_='next')
        page_url = URL + next_btn.find('a')['href'] if next_btn else None
    
    df = pd.DataFrame(quotes_data)
    print(f"মোট {len(df)} quotes সংগ্রহ করা হয়েছে")
    return df

import hashlib, requests

def get_page_hash(url: str) -> str:
    """পেজের unique fingerprint বের করো"""
    content = requests.get(url).content
    return hashlib.md5(content).hexdigest()

def binary_search(arr: list, target) -> int:
    """
    Sorted list-এ target খোঁজো।
    IMPORTANT: list অবশ্যই sorted হতে হবে!
    """
    left, right = 0, len(arr) - 1
    
    while left <= right:
        mid = (left + right) // 2  # মাঝামাঝি index
        
        if arr[mid] == target:
            return mid          # পেয়ে গেছি!
        elif arr[mid] < target:
            left = mid + 1       # ডান দিকে খোঁজো
        else:
            right = mid - 1      # বাম দিকে খোঁজো
    
    return -1

def bubble_sort(arr: list) -> list:
    arr = arr.copy()  # original বদলাবে না
    n = len(arr)
    
    for i in range(n):
        swapped = False
        for j in range(0, n-i-1):
            if arr[j] > arr[j+1]:
                arr[j], arr[j+1] = arr[j+1], arr[j]
                swapped = True
        
        if not swapped:  # কোনো swap না হলে sorted — থামো
            break
    
    return arr

def merge_sort(arr: list) -> list:
    if len(arr) <= 1:
        return arr
    
    # ভাগ করো
    mid = len(arr) // 2
    left  = merge_sort(arr[:mid])
    right = merge_sort(arr[mid:])
    
    # জোড়া দাও
    return _merge(left, right)

def _merge(left: list, right: list) -> list:
    result = []
    i = j = 0
    
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            result.append(left[i]); i += 1
        else:
            result.append(right[j]); j += 1
    
    result.extend(left[i:])   # বাকি যা আছে
    result.extend(right[j:])
    return result

def quick_sort(arr: list) -> list:
    if len(arr) <= 1:
        return arr
    
    pivot = arr[len(arr) // 2]  # মাঝের element pivot
    left   = [x for x in arr if x < pivot]
    middle = [x for x in arr if x == pivot]
    right  = [x for x in arr if x > pivot]
    
    return quick_sort(left) + middle + quick_sort(right)

Python — BFS Web Crawler
from collections import deque
from urllib.parse import urljoin, urlparse
import requests
from bs4 import BeautifulSoup

def bfs_crawler(start_url: str, max_pages: int = 20) -> list:
    """
    BFS দিয়ে website crawl করা।
    Start URL থেকে শুরু করে সব linked page visit।
    """
    visited = set()           # আগে গেছি কিনা track
    queue = deque([start_url])  # FIFO queue — BFS-এর মূল কাঠামো
    base_domain = urlparse(start_url).netloc
    results = []
    
    while queue and len(visited) < max_pages:
        url = queue.popleft()  # সামনে থেকে নাও
        
        if url in visited:
            continue
        
        try:
            resp = requests.get(url, timeout=5)
            soup = BeautifulSoup(resp.content, 'lxml')
            visited.add(url)
            results.append({'url': url, 'title': soup.title.string if soup.title else ''})
            
            # সব লিংক বের করে queue-তে যোগ করো
            for link in soup.find_all('a', href=True):
                full_url = urljoin(url, link['href'])
                # শুধু একই domain-এর links নাও
                if urlparse(full_url).netloc == base_domain:
                    if full_url not in visited:
                        queue.append(full_url)
        
        except:
            pass
    
    return results

from bs4 import BeautifulSoup, Tag

def dfs_extract_text(element, results=None, depth=0) -> list:
    """
    DOM tree-তে DFS দিয়ে সব text বের করা।
    Recursion দিয়ে গভীরে গভীরে যাই।

        for child in element.children:
            dfs_extract_text(child, results, depth + 1)
    
    return results

Exponential Backoff — বুদ্ধিমানের retry

🧒 কিন্ডারগার্টেন ব্যাখ্যা: দরজায় knock করলে কেউ খুললো না। ১ সেকেন্ড অপেক্ষা করে আবার knock। না খুললে ২ সেকেন্ড অপেক্ষা। তারপর ৪ সেকেন্ড, ৮ সেকেন্ড... এভাবে অপেক্ষা দ্বিগুণ হতে থাকে। সার্ভারকে সুযোগ দেওয়া হয় recover করতে।
Python — Exponential Backoff
import time, random

def exponential_backoff(
    func, 
    max_retries: int = 5,
    base_delay: float = 1.0,
    max_delay: float = 60.0,
    jitter: bool = True
):
    """
    যেকোনো function-কে smart retry দিয়ে wrap করো।
    Jitter: random ছোট delay যোগ — thundering herd সমস্যা এড়াতে।
    """
    for attempt in range(max_retries):
        try:
            return func()
        except Exception as e:
            if attempt == max_retries - 1:
                raise  # সব retry শেষ — error তোলো
            
            delay = min(base_delay * (2 ** attempt), max_delay)
            if jitter:
                delay += random.uniform(0, delay * 0.1)
            
            print(f"Attempt {attempt+1} failed: {e}. Retrying in {delay:.1f}s...")
            time.sleep(delay)

# ব্যবহার

    lambda: requests.get('https://example.com', timeout=5)

Cosine Similarity — কতটা একই রকম?

🧒 কিন্ডারগার্টেন ব্যাখ্যা: দুটো তীর কল্পনা করো। যদি একই দিকে পয়েন্ট করে — একই রকম। যদি উল্টো দিকে — একদম আলাদা। Cosine similarity মাপে এই তীরগুলোর মধ্যকার কোণ।
Python — Duplicate Content Detection
from sklearn.metrics.pairwise import cosine_similarity
from sklearn.feature_extraction.text import TfidfVectorizer
import numpy as np

def find_duplicate_articles(articles: list, threshold: float = 0.85) -> list:
    """
    Scraped articles-এ near-duplicate খোঁজা।
    ৮৫% বা বেশি similar হলে duplicate ধরা।
    """
    if len(articles) < 2:
        return []
    
    vectorizer = TfidfVectorizer()
    tfidf = vectorizer.fit_transform(articles)
    sim_matrix = cosine_similarity(tfidf)
    
    duplicates = []
    for i in range(len(articles)):
        for j in range(i+1, len(articles)):
            if sim_matrix[i][j] >= threshold:
                duplicates.append((i, j, sim_matrix[i][j]))
    
    return duplicates

Token Bucket — Scraping Rate Control

🧒 কিন্ডারগার্টেন ব্যাখ্যা: একটা bucket-এ token আছে। প্রতি সেকেন্ডে ১টা নতুন token আসে (সর্বোচ্চ ১০টা থাকতে পারে)। Request করতে হলে একটা token খরচ করতে হয়। Token শেষ হলে অপেক্ষা করো। এটা Scraping-কে polite রাখে!
Python — Token Bucket Rate Limiter
import time
from threading import Lock

class TokenBucket:
    """
    Thread-safe rate limiter scraping-এর জন্য।
    production pipeline-এ ব্যবহার করো।
    """
    
    def __init__(self, rate: float, capacity: int):
        self.rate = rate          # প্রতি সেকেন্ডে tokens
        self.capacity = capacity  # সর্বোচ্চ tokens
        self.tokens = capacity    # শুরুতে ভরা
        self.last_refill = time.time()
        self.lock = Lock()
    
    def consume(self, tokens: int = 1) -> bool:
        with self.lock:
            now = time.time()
            # নতুন tokens যোগ করো
            elapsed = now - self.last_refill
            self.tokens = min(
                self.capacity,
                self.tokens + elapsed * self.rate
            )
            self.last_refill = now
            
            if self.tokens >= tokens:
                self.tokens -= tokens
                return True
            return False
    
    def wait_and_consume(self, tokens: int = 1):
        while not self.consume(tokens):
            time.sleep(0.1)

# ব্যবহার: প্রতি সেকেন্ডে ২টার বেশি request না
limiter = TokenBucket(rate=2.0, capacity=5)
urls = ['https://example.com/1', 'https://example.com/2']

for url in urls:
    limiter.wait_and_consume()  # rate limit মানো
    # resp = requests.get(url)
