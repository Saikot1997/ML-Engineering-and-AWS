Object-Oriented Matplotlib — সবচেয়ে শক্তিশালী পদ্ধতি
📌 এই অধ্যায়ে কী শিখব?
Figure ও Axes object কী, OOP style-এ Matplotlib, subplots, twinx, shared axes — সব বাংলায় একদম নতুনদের জন্য।

দুটি পদ্ধতির মধ্যে পার্থক্য কী?

তুমি আগে হয়তো লিখেছ: plt.plot(x, y) এবং plt.title('গ্রাফ') — এটা pyplot পদ্ধতি। এটা সহজ, কিন্তু সীমিত।

এখন শিখব Object-Oriented (OOP) পদ্ধতি — যেখানে তুমি নিজে Figure এবং Axes তৈরি করো এবং সরাসরি সেগুলো নিয়ন্ত্রণ করো।

উদাহরণ দিয়ে বুঝি: pyplot পদ্ধতিতে তুমি রেস্টুরেন্টে গিয়ে বললে "এক প্লেট ভাত দাও" — তারা নিজেরা সব ঠিক করে দেয়। OOP পদ্ধতিতে তুমি নিজেই রান্নাঘরে যাও, চাল কতটুকু নেবে, কতক্ষণ সিদ্ধ করবে — সব তুমি ঠিক করছ।

চিত্র ১.১: pyplot বনাম OOP — দুটি পদ্ধতির তুলনা

Figure এবং Axes — সবচেয়ে গুরুত্বপূর্ণ ধারণা

Figure = বড় সাদা কাগজ বা ক্যানভাস 🖼️ — এটা হলো পুরো উইন্ডো যেটায় সব গ্রাফ থাকে।

Axes = সেই কাগজের ভেতরে একটি গ্রাফ 📊 — x-axis, y-axis, data সব মিলিয়ে একটি Axes।

একটা Figure-এ একটি বা একাধিক Axes থাকতে পারে — যেমন একটি সংবাদপত্রে (Figure) একাধিক ছবি (Axes) থাকে।

⚠️ সাবধান: "Axes" মানে কিন্তু axis (অক্ষ) নয়! Axes মানে পুরো একটি subplot/graph।

চিত্র ১.২: Figure = বড় ক্যানভাস, Axes = তার ভেতরের প্রতিটি গ্রাফ

add_subplot() — গ্রিডে গ্রাফ সাজানো

fig.add_subplot(numrows, numcols, fignum) — তিনটি সংখ্যা:

  • numrows: কতটি সারি চাই
  • numcols: প্রতি সারিতে কতটি কলাম চাই
  • fignum: এই Axes টি কোন নম্বরে বসবে (বাম-উপর থেকে গণনা)

উদাহরণ: add_subplot(2, 3, 4) = ২ সারি × ৩ কলাম গ্রিডের ৪ নম্বর ঘরে। সংক্ষেপে add_subplot(234) ও লেখা যায়।

চিত্র ১.৩: add_subplot() দিয়ে বিভিন্নভাবে গ্রাফ সাজানো

মূল OOP কোড — লাইন বাই লাইন

python — OOP Matplotlib মূল কাঠামো
import matplotlib.pyplot as plt
import numpy as np

# ━━ ধাপ ১: Figure তৈরি করো ━━
# figsize=(প্রস্থ_ইঞ্চি, উচ্চতা_ইঞ্চি)
fig = plt.figure(figsize=(8, 4))

# ━━ ধাপ ২: Axes যোগ করো ━━
# (১,২,১) = ১ সারি, ২ কলাম, ১ম ঘর  →  বাম গ্রাফ
ax1 = fig.add_subplot(1, 2, 1)
# (১,২,২) = ১ সারি, ২ কলাম, ২য় ঘর  →  ডান গ্রাফ
ax2 = fig.add_subplot(1, 2, 2)

# ━━ ধাপ ৩: গ্রাফ আঁকো ━━
# 'green' = লাইনের রং। বিন্দু: (0,1),(2,3),(3,2)
ax1.plot([0, 2, 3], [1, 3, 2], 'green')

# Histogram: কতবার কোন মান এসেছে
ax2.hist([1, 2, 2, 2, 3, 3, 4, 5, 5])

# ━━ ধাপ ৪: Title ও Labels দাও ━━
# OOP: set_title()  →  pyplot-এ ছিল: plt.title()
ax1.set_title('লাইন গ্রাফ')
ax2.set_title('হিস্টোগ্রাম')
ax1.set_xlabel('x-মান')
ax1.set_ylabel('y-মান')

# tight_layout: গ্রাফগুলো overlap না করে সুন্দরভাবে সাজায়
plt.tight_layout()
plt.show()

Multiple Figures — একসাথে দুটো আলাদা উইন্ডো

একাধিকবার plt.figure() call করলে আলাদা আলাদা Figure তৈরি হয়। এগুলো সম্পূর্ণ স্বাধীন।

python — Multiple Figures

fig1 = plt.figure()

ax1.plot([0, 2, 3], [1, 3, 2], 'green')

fig2 = plt.figure()
ax2 = fig2.add_subplot(111)
ax2.hist([1, 2, 2, 2, 3, 3, 4, 5, 5])

plt.show()  # দুটো উইন্ডো একসাথে দেখাবে

twinx() — একই গ্রাফে দুটো y-axis!

কখনো একটি গ্রাফে দুই ধরনের ডেটা দেখাতে হয় যাদের scale সম্পূর্ণ আলাদা। যেমন তাপমাত্রা (০-৪০°C) এবং বৃষ্টিপাত (০-৫০০মিমি)।

twinx() একই x-axis শেয়ার করে কিন্তু ডান দিকে নতুন y-axis তৈরি করে।

⚠️ দ্বিতীয় curve-এ অবশ্যই আলাদা রং দাও নইলে দুটো একই রঙ হবে এবং confusion হবে।

চিত্র ১.৪: twinx() — বাম ও ডান দুটি y-axis

python — twinx() উদাহরণ
x = np.arange(-10, 10, 0.1)
y1 = x**2   # x-এর বর্গ (মান ০ থেকে ১০০)

# বাম y-axis: নীল curve

plt.subplots() — এক লাইনে সব!

💡 শর্টকাট: plt.subplots() vs fig.add_subplot()

plt.subplots(2, 2) = একসাথে Figure এবং ২×২ সব Axes তৈরি।

axes[i, j] দিয়ে নির্দিষ্ট subplot access করো।

sharex=True: সব গ্রাফের x-axis একই range এ থাকবে। sharey=True: y-axis একই।

python — plt.subplots()
# ২×২ গ্রিড, সব গ্রাফে একই x ও y range
fig, axes = plt.subplots(2, 2, sharex=True, sharey=True)

# axes একটা ২D numpy array → axes[সারি, কলাম]
for i in range(2):
    for j in range(2):
        axes[i, j].hist(
            np.random.randn(500),   # ৫০০টি random সংখ্যা
            bins=50,                # ৫০টি ভাগে ভাগ করো
            color='black',
            alpha=0.5              # ৫০% স্বচ্ছ
        )

plt.tight_layout()
plt.show()

# Shared axes উদাহরণ (তারিখ/signal comparison-এ দরকারী)
x = np.arange(0, 10, 0.01)
fig = plt.figure()
ax1 = fig.add_subplot(221);             ax1.plot(x, np.sin(x))
ax2 = fig.add_subplot(222);             ax2.plot(-x, np.cos(x))
ax3 = fig.add_subplot(223, sharex=ax1); ax3.plot(x*2, np.sin(x*2))
ax4 = fig.add_subplot(224, sharex=ax2); ax4.plot(x, np.cos(x))
plt.tight_layout(); plt.show()

s3 = np.random.randn(30).cumsum()

ax1.set_xticks(range(0, 21, 2))

plt.show()

def bubble_sort(arr):
    n = len(arr)
    steps = [arr.copy()]
    for i in range(n):
        swapped = False
        for j in range(0, n-i-1):
            if arr[j] > arr[j+1]:
                arr[j], arr[j+1] = arr[j+1], arr[j]  # swap
                swapped = True
                steps.append(arr.copy())
        if not swapped: break  # আর swap না হলে থামো
    return arr, steps

python — Merge Sort
def merge_sort(arr):
    if len(arr) <= 1: return arr
    mid = len(arr) // 2
    left  = merge_sort(arr[:mid])   # বাম অর্ধেক সাজাও
    right = merge_sort(arr[mid:])   # ডান অর্ধেক সাজাও
    return merge(left, right)        # একত্রিত করো

def merge(left, right):
    result, i, j = [], 0, 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]: result.append(left[i]);  i += 1
        else:                   result.append(right[j]); j += 1
    result.extend(left[i:]); result.extend(right[j:])
    return result

def quick_sort(arr):
    if len(arr) <= 1: return arr
    pivot = arr[len(arr) // 2]  # মাঝেরটা pivot
    left  = [x for x in arr if x < pivot]
    mid   = [x for x in arr if x == pivot]
    right = [x for x in arr if x > pivot]
    return quick_sort(left) + mid + quick_sort(right)

def binary_search(arr, target):
    left, right = 0, len(arr) - 1
    while left <= right:
        mid = (left + right) // 2
        if   arr[mid] == target: return mid          # ✓ পেয়ে গেছি
        elif arr[mid] < target:  left  = mid + 1      # ডানে যাও
        else:                    right = mid - 1     # বামে যাও
    return -1  # পাওয়া যায়নি

python — BFS ও DFS
from collections import deque

graph = {
    'A': ['B', 'C'], 'B': ['A', 'D'],
    'C': ['A', 'E'], 'D': ['B', 'F'],
    'E': ['C', 'F'], 'F': ['D', 'E']
}

# ━━ BFS: Queue (FIFO) ━━
def bfs(graph, start):
    visited, queue, order = set(), deque([start]), []
    while queue:
        node = queue.popleft()  # সামনে থেকে নাও (FIFO)
        if node not in visited:
            visited.add(node); order.append(node)
            queue.extend(n for n in graph[node] if n not in visited)
    return order

# ━━ DFS: Stack/Recursion ━━
def dfs(graph, start, visited=None):
    if visited is None: visited = set()
    visited.add(start); order = [start]
    for n in graph[start]:
        if n not in visited:
            order.extend(dfs(graph, n, visited))
    return order

print("BFS:", bfs(graph, 'A'))   # ['A','B','C','D','E','F']
print("DFS:", dfs(graph, 'A'))   # ['A','B','D','F','E','C']

from collections import Counter

class KNN:
    def __init__(self, k=3):
        self.k = k

    def fit(self, X, y):
        self.X_train = X; self.y_train = y  # শুধু data মনে রাখো

    def predict(self, X):
        return [self._pred(x) for x in X]

    def _pred(self, x):
        # Euclidean distance: √Σ(x-xᵢ)²
        dists = [np.sqrt(np.sum((x-xt)**2)) for xt in self.X_train]
        k_idx = np.argsort(dists)[:self.k]
        k_labels = [self.y_train[i] for i in k_idx]
        return Counter(k_labels).most_common(1)[0][0]

class KMeans:
    def __init__(self, k=3, max_iter=100):
        self.k = k; self.max_iter = max_iter

    def fit(self, X):
        idx = np.random.choice(len(X), self.k, replace=False)
        self.centroids = X[idx].copy()

        for _ in range(self.max_iter):
            labels = self._assign(X)
            new_c  = np.array([X[labels==k].mean(axis=0) for k in range(self.k)])
            if np.allclose(self.centroids, new_c): break
            self.centroids = new_c

        self.labels_ = self._assign(X); return self

    def _assign(self, X):
        dists = np.array([[np.linalg.norm(x-c) for c in self.centroids] for x in X])
        return np.argmin(dists, axis=1)

python — Linear Regression + Residual Analysis
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score

np.random.seed(42)
X_lr = np.random.randn(100, 1)
y_lr = 3.5*X_lr.flatten() + 2.0 + np.random.randn(100)

model = LinearRegression().fit(X_lr, y_lr)
y_pred = model.predict(X_lr)
residuals = y_lr - y_pred  # actual - predicted

print(f"w={model.coef_[0]:.3f}, b={model.intercept_:.3f}")
print(f"MSE={mean_squared_error(y_lr,y_pred):.3f}, R²={r2_score(y_lr,y_pred):.3f}")

fig, axes = plt.subplots(1, 2, figsize=(12, 5))

# Regression line
axes[0].scatter(X_lr, y_lr, alpha=0.6, color='#58a6ff', s=30)
x_line = np.linspace(X_lr.min(), X_lr.max(), 100).reshape(-1,1)
axes[0].plot(x_line, model.predict(x_line), color='#f78166', lw=2)
axes[0].set_title(r'Linear Regression ($R^2$=' + f'{r2_score(y_lr,y_pred):.2f})')

# Residual plot
axes[1].scatter(y_pred, residuals, alpha=0.6, color='#bc8cff', s=30)
axes[1].axhline(0, color='#3fb950', lw=2, linestyle='--')
axes[1].set_xlabel('Predicted'); axes[1].set_ylabel('Residual')
axes[1].set_title('Residual Plot (0-এর কাছে থাকলে ভালো)')

plt.tight_layout(); plt.show()
